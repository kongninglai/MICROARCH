// ============================================================
// COMBINED7: Prefix spam stress test
// Gratuitous 0x66, segment overrides (26/2E/36/3E/64/65),
// and prefixes before 0x0F extended opcodes.
// 0x66 applied where it does NOT change behavior (reg-direct).
// Segment overrides where they do and don't matter.
// DS=0x200->phys 0x2000, ES=0x400->phys 0x5000, SS=0xb00->phys 0x4000
// FS=0xa00->phys 0x5000, GS not initialized (won't use for mem)
// ============================================================

// --- Setup ---
0x0:   ba 00 02 00 00  // mov    edx,0x200
0x5:   8e da           // mov    ds,edx
0x7:   ba 00 04 00 00  // mov    edx,0x400
0xc:   8e c2           // mov    es,edx
0xe:   ba 00 0b 00 00  // mov    edx,0xb00
0x13:  8e d2           // mov    ss,edx
0x15:  ba 00 0a 00 00  // mov    edx,0xa00
0x1a:  8e e2           // mov    fs,edx
0x1c:  bc 00 01 00 00  // mov    esp,0x100
0x21:  bb 00 00 00 00  // mov    ebx,0x0
0x26:  bd 00 00 00 00  // mov    ebp,0x0

// ============================================================
// SECTION 1: Segment overrides on register-direct ops (no effect)
// The segment prefix is ignored for reg-reg instructions.
// ============================================================

// 26 (ES:) before reg-reg ADD — meaningless but must decode
0x2b:  b8 01 00 00 00  // mov    eax,0x1
0x30:  b9 02 00 00 00  // mov    ecx,0x2
0x35:  26 01 c8        // es: add eax,ecx        ; ES prefix ignored, eax=3

// 2E (CS:) before reg-reg AND
0x38:  b8 ff 00 ff 00  // mov    eax,0x00FF00FF
0x3d:  b9 0f 0f 0f 0f  // mov    ecx,0x0F0F0F0F
0x42:  2e 21 c8        // cs: and eax,ecx        ; CS prefix ignored, eax=0x000F000F

// 36 (SS:) before reg-reg OR
0x45:  b8 f0 f0 f0 f0  // mov    eax,0xF0F0F0F0
0x4a:  b9 0f 0f 0f 0f  // mov    ecx,0x0F0F0F0F
0x4f:  36 09 c8        // ss: or eax,ecx         ; SS prefix ignored, eax=0xFFFFFFFF

// 3E (DS:) before reg-reg XOR (using ADD to simulate)
0x52:  be aa aa aa aa  // mov    esi,0xAAAAAAAA
0x57:  bf 55 55 55 55  // mov    edi,0x55555555
0x5c:  3e 01 fe        // ds: add esi,edi        ; DS prefix ignored, esi=0xFFFFFFFF

// 64 (FS:) before reg-reg MOV
0x5f:  b8 de ad be ef  // mov    eax,0xDEADBEEF
0x64:  64 89 c1        // fs: mov ecx,eax        ; FS prefix ignored, ecx=0xDEADBEEF

// 65 (GS:) before reg-reg SBB (with CF=0)
0x67:  b8 00 00 00 00  // mov    eax,0x0
0x6c:  05 00 00 00 00  // add    eax,0x0         ; CF=0
0x71:  ba 10 00 00 00  // mov    edx,0x10
0x76:  be 03 00 00 00  // mov    esi,0x3
0x7b:  65 19 f2        // gs: sbb edx,esi        ; GS prefix ignored, edx=0xD

// ============================================================
// SECTION 2: Segment overrides on memory ops (meaningful)
// Each override changes which physical memory is accessed.
// ============================================================

// DS: (3E) explicit on DS-default — redundant but valid
0x7e:  c7 43 00 11 11 11 11  // mov [ebx+0x00],0x11111111  ; DS:0x00
0x85:  3e 8b 43 00     // ds: mov eax,[ebx+0x00]  ; DS prefix explicit, eax=0x11111111

// ES: (26) override — accesses ES segment (phys 0x5000)
0x89:  26 c7 45 00 22 22 22 22  // es: mov [ebp+0x00],0x22222222
0x91:  26 8b 4d 00     // es: mov ecx,[ebp+0x00]  ; ecx=0x22222222

// SS: (36) override — accesses SS segment (phys 0x4000)
0x95:  bf 80 00 00 00  // mov    edi,0x80
0x9a:  36 c7 07 33 33 33 33  // ss: mov [edi],0x33333333  ; SS:0x80
0xa1:  36 8b 17        // ss: mov edx,[edi]       ; edx=0x33333333

// FS: (64) override — accesses FS segment (phys 0x5000, same PFN as ES)
0xa4:  64 c7 45 10 44 44 44 44  // fs: mov [ebp+0x10],0x44444444
0xac:  64 8b 75 10     // fs: mov esi,[ebp+0x10]  ; esi=0x44444444

// ============================================================
// SECTION 3: 0x66 prefix on 32-bit reg-direct ops (no effect)
// In 32-bit mode, 0x66 switches to 16-bit operand size.
// On reg-direct, it changes the operation. But here we test
// that the prefix is correctly consumed even when stacked
// with segment overrides.
// ============================================================

// 66 + 26 (ES) before ADD reg,reg (16-bit add)
0xb0:  66 b8 ff 7f     // mov    ax,0x7FFF
0xb4:  66 b9 01 00     // mov    cx,0x1
0xb8:  66 26 01 c8     // es: add ax,cx           ; 16-bit add, ax=0x8000

// 26 + 66 (reversed order) before OR reg,reg
0xbc:  66 b8 f0 00     // mov    ax,0x00F0
0xc0:  66 b9 0f 00     // mov    cx,0x000F
0xc4:  26 66 09 c8     // es: or ax,cx            ; 16-bit or, ax=0x00FF

// 36 (SS) + 66 before AND reg,reg
0xc8:  66 b8 ff ff     // mov    ax,0xFFFF
0xcc:  66 b9 0f 0f     // mov    cx,0x0F0F
0xd0:  36 66 21 c8     // ss: and ax,cx           ; 16-bit and, ax=0x0F0F

// 64 (FS) + 66 before MOV reg,reg
0xd4:  66 b8 34 12     // mov    ax,0x1234
0xd8:  64 66 89 c1     // fs: mov cx,ax           ; 16-bit mov, cx=0x1234

// ============================================================
// SECTION 4: Segment overrides + 66 on MEMORY ops
// ============================================================

// 26 66 ADD [mem],reg (ES:, 16-bit)
0xdc:  26 66 c7 45 20 00 00  // es: mov WORD [ebp+0x20],0x0000
0xe3:  66 b8 ff 00     // mov    ax,0x00FF
0xe7:  26 66 01 45 20  // es: add WORD [ebp+0x20],ax  ; ES:[0x20] = 0x00FF

// 36 66 OR [mem],imm16 (SS:, 16-bit)
0xec:  36 66 c7 47 00 00 f0  // ss: mov WORD [edi],0xF000  ; SS:0x80
0xf3:  36 66 81 0f 0f 00     // ss: or WORD [edi],0x000F   ; SS:0x80 = 0xF00F

// 64 AND [mem],imm32 (FS:)
0xf9:  64 c7 45 20 ff ff ff ff  // fs: mov [ebp+0x20],0xFFFFFFFF
0x101: 64 81 65 20 0f 0f 0f 0f  // fs: and [ebp+0x20],0x0F0F0F0F  ; -> 0x0F0F0F0F

// ============================================================
// SECTION 5: Prefixes before 0x0F extended opcodes
// CPU treats 0x0F as a "prefix" internally, so test 66 and

// --- BSF with segment override (reg-reg, override meaningless) ---

0x116: 36 0f bc f1     // ss: bsf esi,ecx        ; SS prefix ignored, esi=4
0x11a: 3e 0f bc f9     // ds: bsf edi,ecx        ; DS prefix ignored, edi=4

0x15f: ba 55 55 55 55  // mov    edx,0x55555555
