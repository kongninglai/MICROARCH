// ============================================================
// COMBINED8: SIB addressing modes, 8/16-bit memory ALU,
//            store-to-load forwarding, segment register moves,
//            accumulator short-form ALU, back-to-back same-line
//            memory access patterns.
// DS=0x200->phys 0x2000, ES=0x400->phys 0x5000, SS=0xb00->phys 0x4000
// SIB restriction: base != 101 (EBP), index != 100 (ESP)
// ============================================================

// --- Setup ---
0x0:   ba 00 02 00 00  // mov    edx,0x200
0x5:   8e da           // mov    ds,edx
0x7:   ba 00 04 00 00  // mov    edx,0x400

0xe:   ba 00 0b 00 00  // mov    edx,0xb00

0x15:  bc 00 01 00 00  // mov    esp,0x100
0x1a:  bb 00 00 00 00  // mov    ebx,0x0
0x1f:  bd 00 00 00 00  // mov    ebp,0x0

// ============================================================
// SECTION 1: SIB addressing — [base + index*scale + disp]
// base=EBX(011), index=ESI(110)/EDI(111)/ECX(001)
// Never base=101(EBP), never index=100(ESP)
// ============================================================

// Init memory
0x24:  c7 03 aa aa aa aa           // mov [ebx],0xAAAAAAAA           ; DS:0x00
0x2a:  c7 43 10 bb bb bb bb        // mov [ebx+0x10],0xBBBBBBBB     ; DS:0x10
0x31:  c7 43 20 cc cc cc cc        // mov [ebx+0x20],0xCCCCCCCC     ; DS:0x20
0x38:  c7 43 40 dd dd dd dd        // mov [ebx+0x40],0xDDDDDDDD     ; DS:0x40

// [ebx + esi*1] — SIB=0x33 (scale=00, index=110=esi, base=011=ebx)
0x3f:  be 10 00 00 00  // mov    esi,0x10
0x44:  8b 04 33        // mov    eax,[ebx+esi*1]  ; DS:0x10 = 0xBBBBBBBB

// [ebx + esi*2] — SIB=0x73 (scale=01, index=110=esi, base=011=ebx)
0x47:  8b 0c 73        // mov    ecx,[ebx+esi*2]  ; DS:0x20 = 0xCCCCCCCC

// [ebx + esi*4] — SIB=0xB3 (scale=10, index=110=esi, base=011=ebx)
0x4a:  8b 14 b3        // mov    edx,[ebx+esi*4]  ; DS:0x40 = 0xDDDDDDDD

// [ebx + edi*1 + disp8] — SIB=0x3B (scale=00, index=111=edi, base=011=ebx)
0x4d:  bf 00 00 00 00  // mov    edi,0x0
0x52:  8b 44 3b 10     // mov    eax,[ebx+edi*1+0x10]  ; DS:0x10 = 0xBBBBBBBB

// [ebx + ecx*1 + disp32] — SIB=0x0B (scale=00, index=001=ecx, base=011=ebx)
0x56:  b9 00 00 00 00  // mov    ecx,0x0
0x5b:  8b 84 0b 20 00 00 00  // mov eax,[ebx+ecx*1+0x20]  ; DS:0x20 = 0xCCCCCCCC

// [ebx + edi*2 + disp8] — SIB=0x7B (scale=01, index=111=edi, base=011=ebx)
0x62:  8b 44 7b 20     // mov    eax,[ebx+edi*2+0x20]  ; edi=0, DS:0x20 = 0xCCCCCCCC

// SECTION 2: 8-bit memory ALU operations
